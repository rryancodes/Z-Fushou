create table public.community_messages (
  message_id text not null,
  channel_id text not null,
  guild_id text not null,
  user_id text not null,
  username text null,
  content text null,
  timestamp timestamp with time zone not null,
  thread_id text null,
  attachments jsonb null default '[]'::jsonb,
  created_at timestamp with time zone not null default now(),
  is_cleaned boolean null default false,
  cleaned_message_id bigint null,
  is_monitored_mention boolean null default false,
  mentioned_user_ids jsonb null default '[]'::jsonb,
  mention_summary text null,
  mention_summary_generated_at timestamp with time zone null,
  constraint community_messages_pkey primary key (message_id),
  constraint fk_cleaned_message foreign KEY (cleaned_message_id) references community_messages_clean (id) on delete set null
) TABLESPACE pg_default;

create index IF not exists idx_community_messages_channel_id on public.community_messages using btree (channel_id) TABLESPACE pg_default;

create index IF not exists idx_community_messages_timestamp on public.community_messages using btree ("timestamp") TABLESPACE pg_default;

create index IF not exists idx_community_messages_cleaning on public.community_messages using btree (is_cleaned, created_at) TABLESPACE pg_default;

create index IF not exists idx_community_messages_mention_summary_pending on public.community_messages using btree (
  is_monitored_mention,
  mention_summary_generated_at,
  created_at
) TABLESPACE pg_default;

create index IF not exists idx_community_messages_created_at on public.community_messages using btree (created_at desc) TABLESPACE pg_default;

create table public.community_messages_clean (
  id bigserial not null,
  message_id text not null,
  channel_id text null,
  user_id text null,
  username text null,
  content text null,
  timestamp timestamp with time zone null,
  created_at timestamp with time zone null default now(),
  semantic_processed_at timestamp with time zone null,
  live_processed_at timestamp with time zone null,
  constraint community_messages_clean_pkey primary key (id),
  constraint community_messages_clean_message_id_key unique (message_id)
) TABLESPACE pg_default;

create index IF not exists idx_community_messages_clean_message_id on public.community_messages_clean using btree (message_id) TABLESPACE pg_default;

create index IF not exists idx_community_messages_clean_channel_timestamp on public.community_messages_clean using btree (channel_id, "timestamp") TABLESPACE pg_default;

create index IF not exists idx_community_messages_clean_semantic on public.community_messages_clean using btree (semantic_processed_at, created_at) TABLESPACE pg_default;

create index IF not exists idx_community_messages_clean_live_processed on public.community_messages_clean using btree (live_processed_at, created_at) TABLESPACE pg_default;
